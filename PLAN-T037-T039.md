2. **Before returning response (after insights generated, before building JSON response)**:
   ```csharp
   // ── Audit persistence (T037) ──
   // Single transaction: upsert address, insert analysis, backfill raw logs
   using (var transaction = await dbContext.Database.BeginTransactionAsync(ct))
   {
       try
       {
           // Upsert property_addresses (ON CONFLICT DO UPDATE)
           var addressEntity = await EnsureAddressAsync(dbContext, normalizedAddress, ct);
           analysis.AddressId = addressEntity.Id;

           // INSERT property_analyses (append-only)
           analysis.RequestIp = clientIp;
           analysis.ApiConsumerId = apiConsumer?.Id ?? Guid.Empty;
           
           dbContext.PropertyAnalyses.Add(analysis);
           await dbContext.SaveChangesAsync(ct);

           // Backfill data_provider_raw_logs with analysis_id
           // WHERE address_id = :addressId AND fetched_at >= :requestStart AND analysis_id IS NULL
           await BackfillRawLogsAsync(dbContext, addressEntity.Id, analysis.Id, requestStart, ct);

           await transaction.CommitAsync(ct);

           logger.LogInformation(
               "Analysis persisted. AnalysisId={AnalysisId} AddressId={AddressId} Composite={Composite}",
               analysis.Id,
               analysis.AddressId,
               analysis.CompositeScore);
       }
       catch (Exception ex)
       {
           await transaction.RollbackAsync(ct);
           logger.LogError(
               ex,
               "Failed to persist analysis. CorrelationId={CorrelationId} Continuing without audit.",
               correlationId);
           // Do NOT throw — return the response anyway (audit loss is non-fatal)
       }
   }
   ```

3. **Implement helper methods** (at the end of the `AnalyzeEndpoint` class):

   ```csharp
   private static async Task<PropertyAddress> EnsureAddressAsync(
       PropertyIntelligenceDbContext dbContext,
       PropertyAddress normalizedAddress,
       CancellationToken ct)
   {
       // ON CONFLICT normalized_address DO NOTHING, then fetch the actual ID
       var existing = await dbContext.PropertyAddresses
           .FirstOrDefaultAsync(a => a.NormalizedAddress == normalizedAddress.NormalizedAddress, ct);

       if (existing is not null)
       {
           return existing;
       }

       normalizedAddress.CreatedAt = DateTime.UtcNow;
       dbContext.PropertyAddresses.Add(normalizedAddress);
       await dbContext.SaveChangesAsync(ct);
       return normalizedAddress;
   }

   private static async Task BackfillRawLogsAsync(
       PropertyIntelligenceDbContext dbContext,
       Guid addressId,
       Guid analysisId,
       DateTime requestStart,
       CancellationToken ct)
   {
       // Raw logs created during enrichment but not yet linked to analysis_id
       var logsToUpdate = await dbContext.DataProviderRawLogs
           .Where(log => log.AddressId == addressId
                      && log.FetchedAt >= requestStart
                      && log.AnalysisId == null)
           .ToListAsync(ct);

       foreach (var log in logsToUpdate)
       {
           log.AnalysisId = analysisId;
       }

       if (logsToUpdate.Count > 0)
       {
           await dbContext.SaveChangesAsync(ct);
       }
   }
   ```

4. **Capture `requestStart` timestamp** (at the beginning of `HandleAsync`):
   ```csharp
   var requestStart = DateTime.UtcNow;  // Capture before any async work
   ```

5. **Update the error handling for consistency** (StatusCode 503 case):
   - When all critical providers fail: Log with correlation ID before returning 503
   - Include `analysis_id` null in log if no analysis was generated

**Testing**:

## Orchestration Strategy: Pi-Subagents Parallel Execution

### Worktree Setup

```bash
# In main repo:
cd /home/helio/repos/property-intelligence

# Create worktrees for each task
git worktree add ../pi-t037 main
git worktree add ../pi-t038 main
git worktree add ../pi-t039 main

# Verify:
git worktree list
```

### Subagent Delegation Command

```javascript
// T037 + T038 run in parallel; T039 waits for both
pi_messenger({ 
  action: "plan",
  prompt: "Complete T037–T039 per /home/helio/repos/property-intelligence/PLAN-T037-T039.md"
})

// Then:
subagent({
  tasks: [
    {
      agent: "worker",
      task: "Implement T037 (Audit Persistence) per PLAN-T037-T039.md. Write DbContext injection + transaction logic in AnalyzeEndpoint.cs. Commit to t037 branch."
    },
    {
      agent: "worker",
      task: "Implement T038 (Mock Config) per PLAN-T037-T039.md. Create appsettings.Mock.json with 8 mock providers. Commit to t038 branch."
    }
  ],
  worktree: true,
  async: true
})

// Wait for both to complete, then:
subagent({
  agent: "worker",
  task: "Implement T039 (DI Wiring) per PLAN-T037-T039.md. Integrate T037 + T038 outputs into Program.cs. Commit to t039 branch. Verify dotnet build succeeds."
})
```
